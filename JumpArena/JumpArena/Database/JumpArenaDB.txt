--LichSuHoatDong
CREATE TABLE AuditLog (
    LogID INT IDENTITY(1,1) PRIMARY KEY,
    InvoiceID INT NULL,
    UserID INT NULL,
    ActionType VARCHAR(50),
    Timestamp DATETIME NOT NULL,
    Changes NVARCHAR(MAX)
);

--LoaiSanPham
CREATE TABLE Category (
    CategoryID INT IDENTITY(1,1) PRIMARY KEY,
    CategoryName VARCHAR(100) NOT NULL
);

--KhachHang
CREATE TABLE Customer (
    CustomerID INT IDENTITY(1,1) PRIMARY KEY,
    PhoneNumber VARCHAR(15) NOT NULL,
    LastName VARCHAR(50) NOT NULL,
    MiddleName VARCHAR(50),
    FirstName VARCHAR(50) NOT NULL,
    BirthDate DATETIME NULL,
    Address VARCHAR(255),
    CompanyName VARCHAR(255),
    CompanyAddress VARCHAR(255),
    TaxCode VARCHAR(20),
    LegalRepresentative VARCHAR(100)
);

--Voucher
CREATE TABLE Voucher (
    VoucherID INT IDENTITY(1,1) PRIMARY KEY,
    VoucherCode VARCHAR(50) NOT NULL,
    Description VARCHAR(255),
    DiscountType VARCHAR(20),
    DiscountValue DECIMAL(18,0) NOT NULL,
    MinimumOrderValue DECIMAL(18,0) NOT NULL,
    StartDate DATETIME,
    EndDate DATETIME
);

--VaiTro
CREATE TABLE [Role] (
    RoleID INT IDENTITY(1,1) PRIMARY KEY,
    RoleName VARCHAR(50) NOT NULL
);

--NguoiDung
CREATE TABLE [User] (
    UserID INT IDENTITY(1,1) PRIMARY KEY,
    FullName VARCHAR(100) NOT NULL,
    BirthDate DATETIME NULL,
    Address VARCHAR(255),
    Username VARCHAR(50) NOT NULL,
    Password VARCHAR(255) NOT NULL,
    RoleID INT NULL,
    FOREIGN KEY (RoleID) REFERENCES Role(RoleID)
);

--SanPham
CREATE TABLE [Product] (
    ProductID INT IDENTITY(1,1) PRIMARY KEY,
    ProductName VARCHAR(150) NOT NULL,
    Description VARCHAR(500),
    Price DECIMAL(18,0) NOT NULL,
    StockQuantity INT NOT NULL,
    CategoryID INT NULL,
    FOREIGN KEY (CategoryID) REFERENCES Category(CategoryID)
);

--HoaDon
CREATE TABLE Invoice (
    InvoiceID INT IDENTITY(1,1) PRIMARY KEY,
    CreatedDate DATETIME NOT NULL,
    TotalOriginalAmount DECIMAL(18,0) NOT NULL,
    DiscountAmount DECIMAL(18,0) NOT NULL,
    Status VARCHAR(50),
    PaymentStatus VARCHAR(50),
    CustomerID INT NULL,
    UserID INT NULL,
    VoucherID INT NULL,
    FOREIGN KEY (CustomerID) REFERENCES Customer(CustomerID),
    FOREIGN KEY (UserID) REFERENCES [User](UserID),
    FOREIGN KEY (VoucherID) REFERENCES Voucher(VoucherID)
);

--PhuonhThucThanhToan
CREATE TABLE PaymentMethod (
    PaymentMethodID INT IDENTITY(1,1) PRIMARY KEY,
    MethodName VARCHAR(100) NOT NULL,
    IsDigital BIT NOT NULL
);

--ChiTietHoaDon
CREATE TABLE InvoiceDetail (
    InvoiceDetailID INT IDENTITY(1,1) PRIMARY KEY,
    Quantity INT NOT NULL,
    PriceAtPurchase DECIMAL(18,0) NOT NULL,
    InvoiceID INT NOT NULL,
    ProductID INT NOT NULL,
    FOREIGN KEY (InvoiceID) REFERENCES Invoice(InvoiceID),
    FOREIGN KEY (ProductID) REFERENCES Product(ProductID)
);

--MaCheckin
CREATE TABLE CheckinCode (
    CheckinID INT IDENTITY(1,1) PRIMARY KEY,
    Code VARCHAR(20) NOT NULL,
    CheckinTime DATETIME NULL,
    CheckoutTime DATETIME NULL,
    ExpiryDate DATETIME NOT NULL,
    Status VARCHAR(50),
    InvoiceDetailID INT NULL,
    CustomerID INT NULL,
    FOREIGN KEY (InvoiceDetailID) REFERENCES InvoiceDetail(InvoiceDetailID),
    FOREIGN KEY (CustomerID) REFERENCES Customer(CustomerID)
);

--ThanhToan
CREATE TABLE PaymentTransaction (
    PaymentTransactionID INT IDENTITY(1,1) PRIMARY KEY,
    Amount DECIMAL(18,0) NOT NULL,
    PaymentDate DATETIME NOT NULL,
    TransactionCode VARCHAR(100),
    Note VARCHAR(255),
    InvoiceID INT NOT NULL,
    PaymentMethodID INT NOT NULL,
    FOREIGN KEY (InvoiceID) REFERENCES Invoice(InvoiceID),
    FOREIGN KEY (PaymentMethodID) REFERENCES PaymentMethod(PaymentMethodID)
);
